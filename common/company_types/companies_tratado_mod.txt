company_east_indies_trading_company = {
	icon = "gfx/interface/icons/company_icons/basic_shipyards.dds"
	background = "gfx/interface/icons/company_icons/company_backgrounds/comp_illu_harbor_shipbuilding.dds"
	
	building_types = {
		building_trade_center
		building_port
	}

	extension_building_types = {
		building_shipyard
	}
	
	possible_prestige_goods = {
		prestige_good_generic_merchant_marine
	}

	prestige_goods_trigger = {
		custom_tooltip = {
			text = je_prestige_goods_merchant_marine_complete_tt
			has_variable = prestige_good_generic_merchant_marine_var
		}
	}
	
	potential = {
		is_in_geographic_region = geographic_region_han_homelands
	}

    attainable = {
	}

	possible = {
        any_country = {
            country_rank >= rank_value:great_power
        }
		any_scope_state = {
            is_in_geographic_region = geographic_region_han_homelands
			is_treaty_port = yes
            any_scope_building = {
				is_building_type = building_trade_center
				level >= 5
			}
		}
		NOT = {
			OR = {
				has_company = company_type:company_augustine_heard
				has_company = company_type:company_remi_schmidt
				has_company = company_type:company_gibb_livingston
			}
		}
	}
	
	prosperity_modifier = {
		state_trade_advantage_mult = 0.1
        country_max_companies_add = 1
	}
	
	ai_weight = {
		value = 2
	}
}
