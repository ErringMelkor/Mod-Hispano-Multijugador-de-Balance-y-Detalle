amendment_provincial_governors = {
	
	parent = law_appointed_bureaucrats

	allowed_laws = {
		law_appointed_bureaucrats
	}

	modifier = {
		state_tax_capacity_mult = -0.15
		country_government_wages_mult = -0.25
	}

	on_activate = {	}

	possible = {
		country_has_primary_culture = cu:manchu
	}

	would_sponsor = {
		OR = {
			is_interest_group_type = ig_landowners
			is_interest_group_type = ig_intelligentsia
		}
	}

	ai_will_revoke = {
		NOT = {
			any_interest_group = {
				is_in_government = yes
				OR = {
					is_interest_group_type = ig_landowners
					is_interest_group_type = ig_intelligentsia
				}
			}
		}
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
		has_technology_researched = central_archives
	}

	amendment_activism_multiplier = 0.75
}
