je_questione_della_lingua = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	# Italian Buttons



	is_shown_when_inactive = {
		OR = {
			AND = {
				OR = {
					country_has_primary_culture = cu:north_italian
					country_has_primary_culture = cu:central_italian
					country_has_primary_culture = cu:south_italian

				}
				OR = {
					any_scope_state = {
						state_region = {
							is_homeland = cu:north_italian
						}
					}
					any_scope_state = {
						state_region = {
							is_homeland = cu:central_italian
						}
					}
					any_scope_state = {
						state_region = {
							is_homeland = cu:south_italian
						}
					}
				}
				
			}
		}
		NOT = {
			has_variable = je_vernacular_policy_var
		}
	}

	possible = {
		trigger_if = {
			limit = {
				c:ITA ?= this
			}
			any_scope_state = {
				state_region = {
					is_homeland = cu:north_italian
				}
			}
		}
	}

	immediate = {
		set_variable = je_vernacular_policy_var
		if = {
			limit = {
				c:ITA ?= this
			}
			cu:north_italian ={
				save_scope_as = official_culture_scope1
			}
		}
		trigger_event = {
			id = questione_della_lingua_events.1
			popup = yes
		}
	}

	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = chose_assimilation_var
				}
				clear_variable_list = vernacular_policy_assimilation_states1
				clear_variable_list = vernacular_policy_assimilation_states2
				clear_variable_list = vernacular_policy_assimilation_states3
				every_scope_state = {
					limit = {
						state_region = {
							is_homeland = cu:north_italian
						}
						culture_percent_state = {
							target = cu:north_italian
							value < 0.6
						}
					}
					root = {
						add_to_variable_list = {
							name = vernacular_policy_assimilation_states1
							target = prev
						}
					}
				}
				every_scope_state = {
					limit = {
						state_region = {
							is_homeland = cu:central_italian
						}
						culture_percent_state = {
							target = cu:central_italian
							value < 0.6
						}
					}
					root = {
						add_to_variable_list = {
							name = vernacular_policy_assimilation_states2
							target = prev
						}
					}
				}
				every_scope_state = {
					limit = {
						state_region = {
							is_homeland = cu:south_italian
						}
						culture_percent_state = {
							target = cu:south_italian
							value < 0.6
						}
					}
					root = {
						add_to_variable_list = {
							name = vernacular_policy_assimilation_states3
							target = prev
						}
					}
				}
			}
		}
	}

	complete = {
		AND = {
			any_scope_state = {
				filter = {
					state_region = {
						is_homeland = cu:north_italian
					}
				}
				culture_percent_state = {
					target = cu:north_italian
					value >= 0.6
				}
				custom_tooltip = {
					text = VERNACULAR_POLICY_ASSIMILATION_STATES_TT
				}
				percent = 1
			}
			any_scope_state = {
				filter = {
					state_region = {
						is_homeland = cu:central_italian
					}
				}
				culture_percent_state = {
					target = cu:central_italian
					value >= 0.6
				}
				custom_tooltip = {
					text = VERNACULAR_POLICY_ASSIMILATION_STATES_TT
				}
				percent = 1
			}
			any_scope_state = {
				filter = {
					state_region = {
						is_homeland = cu:south_italian
					}
				}
				culture_percent_state = {
					target = cu:south_italian
					value >= 0.6
				}
				custom_tooltip = {
					text = VERNACULAR_POLICY_ASSIMILATION_STATES_TT
				}
				percent = 1
			}
		}
	}

	on_complete = {
		clear_variable_list = vernacular_policy_assimilation_states1
		clear_variable_list = vernacular_policy_assimilation_states2
		clear_variable_list = vernacular_policy_assimilation_states3
		if = {
			limit = {
				has_variable = chose_assimilation_var
			}
			trigger_event = {
				id = questione_della_lingua_events.2
				popup = yes
			}
		}
	}

	invalid = {
		has_variable = chose_to_not_care_var
	}

	weight = 10000

		
	should_be_pinned_by_default_uninvolved_or_context = yes

}
