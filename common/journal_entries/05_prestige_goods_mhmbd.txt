je_prestige_good_specialty_sugar = {
    icon = "gfx/interface/icons/goods_icons/mhmbd_goods_icons/specialty_sugar.dds"

	group = je_group_technology

    activation_description = {
		always = yes
	}

    is_shown_when_inactive = {
        any_company = {
            can_potentially_produce_prestige_goods = prestige_good_specialty_sugar
            is_producing_prestige_goods = no
            NOT = {
                has_variable = prestige_good_specialty_sugar_var
            }
        }
    }

    possible = {
        any_company = {
            can_potentially_produce_prestige_goods = prestige_good_specialty_sugar
            is_producing_prestige_goods = no
            company_is_prosperous = yes
            NOT = {
                custom_tooltip = {
                    text = not_prestige_good_specialty_sugar_var_tt
                    has_variable = prestige_good_specialty_sugar_var
                }
            }
        }
    }

    immediate = {
        random_company = {
            limit = {
                can_potentially_produce_prestige_goods = prestige_good_specialty_sugar
                is_producing_prestige_goods = no
                company_is_prosperous = yes
                NOT = {
                    has_variable = prestige_good_specialty_sugar_var
                }
            }
            save_scope_as = relevant_company
        }
		set_variable = {
			name = prestige_sugar_timer_var
			value = 0
		}
        g:sugar = {
            save_scope_as = relevant_goods
        }
    }

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            trigger_if = {
                limit = {
                    has_variable = prestige_sugar_timer_var
                }
                scope:journal_entry = {
                    is_goal_complete = yes
                }
            }
            trigger_if = {
                limit = {
                    exists = scope:relevant_company
                }
                scope:relevant_company ?= {
                    company_is_prosperous = yes
                }
            }
            trigger_if = {
                limit = {
                    exists = scope:relevant_goods
                }
                custom_tooltip = {
                    text = relevant_goods_production_rank_tt
                    "goods_production_rank(scope:relevant_goods)" <= 3
                }
            }
        }
        effect = {
            if = {
                limit = {
                    exists = scope:relevant_company
                }
                custom_tooltip = {
                    text = je_prestige_good_specialty_sugar_reward_tt
                    scope:relevant_company ?= {
                        set_variable = prestige_good_specialty_sugar_var
                    }
                }
            }
            post_notification = company_prestige_goods_production_activated
            je_prestige_good_specialty_sugar_cleanup = yes
        }
    }

    on_monthly_pulse = {
        effect = {
            rebind_prestige_good_company = {
                GOODS = prestige_good_specialty_sugar
                UNLOCK_VAR = prestige_good_specialty_sugar_var
                TIMER_VAR = prestige_sugar_timer_var
            }
            if = {
                limit = {
                    scope:relevant_company ?= {
                        company_is_prosperous = yes
                    }
                    "goods_production_rank(scope:relevant_goods)" <= 3
                }
                change_variable = { name = prestige_sugar_timer_var add = 1 }
            }
        }
    }

    can_deactivate = yes

	current_value = {
		value = root.var:prestige_sugar_timer_var
	}

	goal_add_value = {
		add = prestige_good_journal_entry_months
	}

	progressbar = yes

    weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
