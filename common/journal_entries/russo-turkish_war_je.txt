						country_or_subject_owns_entire_state_region = STATE_WALLACHIA
						any_scope_state = {
							state_region = s:STATE_WALLACHIA
							is_incorporated = yes
						}
					}
					AND = {
						country_or_subject_owns_entire_state_region = STATE_MOLDAVIA

							state_region = s:STATE_MOLDAVIA

				}
				s:STATE_MOLDAVIA = {
					remove_claim = ROOT
				}
			}
			if = {

				}
				s:STATE_WALLACHIA = {
